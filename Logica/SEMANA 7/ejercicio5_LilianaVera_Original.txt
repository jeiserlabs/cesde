DATOS DE ENTRADA 
	Tipo_de_trabajador=""
        Nombre_trabajador=""
	Horas_trabajadas=0
	Salario_basico_hora=0
	Deducciones=0
	Bonificaciones=0

PROCESO
  si (Tipo_de_trabajador="FIJO" o Tipo_de_trabajador="fijo")
      Leer Horas_trabajadas
      Leer Salario_basico_hora
      Leer Deducciones
      Leer Bonificaciones
      
  sino
     si (Tipo_de_trabajador="TEMPORAL" O Tipo_de_trabajador="temporal"
         Leer Horas_trabajadas
         Salario_basico_hora=6000
         Bonificaciones=0
         Deducciones=0
     sino
         Escribir "Error.... Tipo de trabajador incorrecto!!!"
     finsi
  finsi
     Salario_Neto = Horas_trabajadas * Salario_basico_hora + Bonificaciones - Deducciones


DATOS DE SALIDA
    Tipo_de_trabajador
    Nombre_trabajador
    Horas_trabajadas
    Salario_basico_hora
    Deducciones
    Bonificaciones
    Salario_Neto

ALGORITMO
   Tipo_de_trabajador=""

      Escribir "Cantidad de horas trabajadas: "
      Leer Horas_trabajadas
      Escribir "Valor hora de trabajo: "
      Leer Salario_basico_hora
      Escribir "Valor deducciones: "
      Leer Deducciones
      Escribir "Valor de las bonificaciones: "
      Leer Bonificaciones
  sino
     si (Tipo_de_trabajador="TEMPORAL" O Tipo_de_trabajador="temporal"
         Escribir "Cantidad de horas trabajadas: "
         Leer Horas_trabajadas
         Salario_basico_hora=6000
         Bonificaciones=0
         Deducciones=0
     sino
         Escribir "Error.... Tipo de trabajador incorrecto!!!"
     finsi
  finsi
     Salario_Neto = Horas_trabajadas * Salario_basico_hora + Bonificaciones - Deducciones

   //DATOS DE SALIDA -  MOSTRAR
   Escribir "--------------- INFORME ---------------"
